\chapter{Phần mở đầu}

\section{Tên đề tài}
{Đề tài 6 - Chuyển tiền Ngân hàng: Two-Phase Commit (2PC).}

Đề tài mô phỏng bài toán chuyển tiền liên ngân hàng trong hệ thống phân tán, yêu cầu giao dịch phải đảm bảo tính nguyên tử:
\begin{itemize}
    \item Hoặc trừ tiền ở tài khoản nguồn {và} cộng tiền ở tài khoản đích đều thành công.
    \item Hoặc toàn bộ giao dịch bị hủy, không để xảy ra tình trạng lệch dữ liệu.
\end{itemize}

\section{Phạm vi và đối tượng nghiên cứu}

\subsection{Đối tượng nghiên cứu}

Đối tượng nghiên cứu của đề tài là hệ thống cơ sở dữ liệu phân tán trong lĩnh vực ngân hàng, cụ thể là bài toán chuyển tiền giữa các tài khoản thuộc các chi nhánh khác nhau. Hệ thống được xây dựng nhằm mô phỏng hoạt động của các ngân hàng có nhiều site (chi nhánh) hoạt động độc lập nhưng vẫn đảm bảo tính nhất quán dữ liệu trên toàn hệ thống.

Bên cạnh đó, đề tài tập trung nghiên cứu giao thức Two-Phase Commit (2PC) để xử lý các giao dịch phân tán, đảm bảo các tính chất ACID, đặc biệt là tính nguyên tử (Atomicity) trong quá trình chuyển tiền giữa các site.

\subsection{Phạm vi nghiên cứu}

Phạm vi của đề tài được giới hạn trong các nội dung sau:

\begin{itemize}
    \item Mô phỏng hệ thống gồm từ 2 đến 3 site ngân hàng, mỗi site quản lý một tập dữ liệu riêng biệt.
    \item Xây dựng chức năng chuyển tiền giữa các tài khoản thuộc các site khác nhau thông qua giao dịch phân tán.
    \item Áp dụng giao thức Two-Phase Commit (2PC) để đảm bảo tính toàn vẹn và nhất quán dữ liệu.
    \item Sử dụng hệ quản trị cơ sở dữ liệu SQL Server kết hợp với cơ chế Linked Server để kết nối giữa các site.
    \item Triển khai các stored procedure để thực hiện truy vấn và giao dịch phân tán.
\end{itemize}

Ngoài ra, đề tài không tập trung vào các vấn đề sau:

\begin{itemize}
    \item Không triển khai trên môi trường thực tế với quy mô lớn (production system).
    \item Không xử lý các nghiệp vụ ngân hàng phức tạp như tín dụng, đầu tư hoặc quản lý rủi ro.
    \item Không tối ưu hiệu năng ở mức cao hoặc xử lý đồng thời số lượng lớn giao dịch.
\end{itemize}

\section{Mục tiêu}
\subsection{Mục tiêu tổng quát}
Xây dựng hệ thống chuyển tiền liên ngân hàng có khả năng chịu lỗi, sử dụng giao thức Two-Phase Commit trên nhiều CSDL độc lập.

\subsection{Mục tiêu cụ thể}
\begin{itemize}
    \item Triển khai được giao dịch phân tán giữa ít nhất 2 site ngân hàng.
    \item Hỗ trợ các tình huống lỗi thường gặp: fail ở Phase 1, fail ở Phase 2, timeout, coordinator crash, participant crash.
    \item Có cơ chế \textit{recovery} từ log để xử lý giao dịch treo.
    \item Có cơ chế \textit{idempotency} để tránh double-submit.
    \item Có bộ test matrix TC01-TC16 và CI để tự động hóa kiểm thử.
\end{itemize}

\section{Phạm vi hệ thống}
\subsection{Phạm vi thực hiện}
\begin{itemize}
    \item {Backend:} Flask (Python), API REST.
    \item {Database:} 3 MySQL site (bank1, bank2, bank3) chạy qua Docker Compose.
    \item {Giao thức giao dịch phân tán:} XA transaction + 2PC do coordinator điều phối.
    \item {Frontend:} HTML/CSS/JavaScript với màn hình monitor giao dịch.
    \item {Mô phỏng lỗi mạng:} Toxiproxy (cấu hình qua cổng 8666).
\end{itemize}

\subsection{Ngoài phạm vi}
\begin{itemize}
    \item Chưa triển khai xác thực/phân quyền mức production (JWT/OAuth).
    \item Chưa triển khai replication cấp CSDL để HA thực sự.
    \item Chưa tối ưu cho tải cao quy mô lớn như production banking.
\end{itemize}

\section{Hệ thống đang ở mức demo hay production?}
Hệ thống trong đề tài này ở mức {demo học thuật nâng cao (production-like)}:
\begin{itemize}
    \item {Đạt được:} có đầy đủ flow 2PC, recovery, compensation, idempotency, monitor, CI test.
    \item {Chưa đạt production đầy đủ:} chưa có cluster HA thực sự, chưa hardening security, chưa có observability và disaster recovery cấp doanh nghiệp.
\end{itemize}

\chapter{Cơ sở lý thuyết}

\section{CSDL phân tán là gì}
CSDL phân tán là hệ thống dữ liệu được lưu trữ trên nhiều nút (site), có thể phân bố địa lý khác nhau, nhưng đối với người dùng và ứng dụng nó vẫn được nhìn như một CSDL thống nhất.

{Lợi ích chính:}
\begin{itemize}
    \item Tăng khả năng mở rộng và phân bố tải.
    \item Tăng khả năng sẵn sàng và chịu lỗi cục bộ.
    \item Gắn dữ liệu với nơi phát sinh nghiệp vụ (theo chi nhánh, khu vực).
\end{itemize}

{Thách thức chính:}
\begin{itemize}
    \item Đảm bảo nhất quán dữ liệu giữa các site.
    \item Đồng bộ và xử lý xung đột.
    \item Quản lý giao dịch phân tán và cơ chế recovery.
\end{itemize}

\section{Phân mảnh dữ liệu}

\subsection{Horizontal Fragmentation}
Là chia bảng theo {dòng} dựa trên điều kiện (predicate). Mỗi site giữ một tập dòng.

{Ví dụ:}
\begin{itemize}
    \item Site A giữ tài khoản có account\_number thuộc nhóm A.
    \item Site B giữ tài khoản có account\_number thuộc nhóm B.
\end{itemize}

\subsection{Vertical Fragmentation}
Là chia bảng theo {cột}. Mỗi site giữ một nhóm cột phù hợp nghiệp vụ.

{Ví dụ:}
\begin{itemize}
    \item Site 1 giữ cột nhận dạng và giao dịch.
    \item Site 2 giữ cột thông tin mở rộng/phân tích.
\end{itemize}

\subsection{Hybrid Fragmentation}
Là kết hợp horizontal và vertical fragmentation để đạt hiệu quả cao hơn theo từng nhóm nghiệp vụ.

\section{Replication}

\subsection{Snapshot Replication}
Sao chép dữ liệu theo từng đợt, phù hợp dữ liệu ít thay đổi hoặc chấp nhận độ trễ cập nhật.

\subsection{Transactional Replication}
Sao chép theo luồng giao dịch gần-thời-gian-thực, phù hợp hệ thống cần độ trễ thấp và nhất quán cao hơn.

{Nhận xét đối với đề tài:}
Trong project hiện tại, trọng tâm là atomicity của giao dịch chuyển tiền, do đó 2PC được ưu tiên. Replication cấp CSDL chưa là tâm điểm triển khai.

\section{Giao dịch phân tán}

\subsection{ACID}
\begin{itemize}
    \item {Atomicity:} tất cả hoặc không gì cả.
    \item {Consistency:} dữ liệu luôn hợp lệ sau giao dịch.
    \item {Isolation:} giao dịch đồng thời không làm hỏng nhau.
    \item {Durability:} đã commit thì phải bền vững.
\end{itemize}

\subsection{Two-Phase Commit (2PC)}
2PC có 2 pha:
\begin{enumerate}
    \item {Phase 1 - Prepare:} coordinator yêu cầu tất cả participant chuẩn bị, participant trả lời YES/NO.
    \item {Phase 2 - Commit/Abort:} nếu tất cả YES thì commit, ngược lại rollback.
\end{enumerate}

Trong đề tài, 2PC được triển khai thông qua XA command của MySQL (\texttt{XA START}, \texttt{XA END}, \texttt{XA PREPARE}, \texttt{XA COMMIT}, \texttt{XA ROLLBACK}).

\chapter{Phân tích hệ thống}

\section{Phát biểu bài toán}

Trong hệ thống ngân hàng hiện đại, các chi nhánh thường được triển khai dưới dạng các hệ cơ sở dữ liệu độc lập (distributed sites), mỗi site quản lý dữ liệu của một tập khách hàng riêng. Khi phát sinh nhu cầu chuyển tiền giữa các tài khoản thuộc hai chi nhánh khác nhau, hệ thống cần thực hiện một giao dịch phân tán (distributed transaction) liên quan đến nhiều site.

Cụ thể, xét bài toán chuyển tiền từ tài khoản thuộc ngân hàng A (Site A) sang tài khoản thuộc ngân hàng B (Site B). Giao dịch này bao gồm hai thao tác chính:
\begin{itemize}
    \item Trừ tiền từ tài khoản nguồn tại Site A.
    \item Cộng tiền vào tài khoản đích tại Site B.
\end{itemize}

Do hai thao tác này diễn ra trên hai hệ thống cơ sở dữ liệu khác nhau, hệ thống phải đảm bảo các yêu cầu nghiêm ngặt về tính toàn vẹn và nhất quán dữ liệu.

Các yêu cầu đặt ra cho hệ thống bao gồm:

\begin{itemize}
    \item \textbf{Tính nguyên tử (Atomicity):} Giao dịch phải được thực hiện toàn bộ hoặc không thực hiện gì cả. Không được xảy ra tình trạng tiền bị trừ ở Site A nhưng không được cộng ở Site B, hoặc ngược lại.
    
    \item \textbf{Tính nhất quán (Consistency):} Sau khi giao dịch hoàn tất, dữ liệu tại các site phải luôn ở trạng thái hợp lệ, không vi phạm các ràng buộc (ví dụ: số dư tài khoản không được âm).
    
    \item \textbf{Khả năng chịu lỗi (Fault Tolerance):} Hệ thống phải có khả năng xử lý các tình huống lỗi như mất kết nối mạng, timeout, hoặc một site bị crash trong quá trình thực hiện giao dịch. Trong các trường hợp này, hệ thống phải có cơ chế rollback hoặc phục hồi phù hợp để đảm bảo dữ liệu không bị sai lệch.
    
    \item \textbf{Theo dõi và truy vết giao dịch:} Hệ thống cần lưu lại trạng thái của giao dịch theo từng giai đoạn (phase), bao gồm các bước như chuẩn bị (prepare), cam kết (commit) hoặc hủy (abort), nhằm phục vụ cho việc kiểm tra, debug và phục hồi khi có sự cố.
    
    \item \textbf{Tính đồng bộ giữa các site:} Các site tham gia giao dịch cần phối hợp chặt chẽ thông qua một cơ chế điều phối (coordinator) để đảm bảo tất cả cùng đạt trạng thái nhất quán cuối cùng.
\end{itemize}

Để giải quyết bài toán trên, đề tài áp dụng giao thức Two-Phase Commit (2PC), trong đó một site đóng vai trò điều phối (coordinator) và các site còn lại đóng vai trò tham gia (participants). Giao thức này giúp đảm bảo rằng tất cả các site hoặc cùng commit giao dịch, hoặc cùng rollback, từ đó duy trì tính toàn vẹn dữ liệu trong môi trường phân tán.

\section{Mô hình nghiệp vụ tổng quát}
\begin{enumerate}
    \item Người dùng đăng nhập.
    \item Nhập thông tin chuyển tiền.
    \item Hệ thống tìm tài khoản đích.
    \item Coordinator thực hiện 2PC.
    \item Trả kết quả, lưu log phase và hóa đơn giao dịch.
    \item Khi có lỗi/treo, hệ thống có recovery tự động hoặc thủ công.
\end{enumerate}

\section{Use case}
\subsection{Tác nhân}
\begin{itemize}
    \item Người dùng (khách hàng).
    \item Hệ thống coordinator (backend Flask).
    \item Site ngân hàng A/B (MySQL participant).
\end{itemize}

\subsection{Danh sách use case}
\begin{itemize}
    \item Đăng nhập.
    \item Tra cứu tài khoản.
    \item Chuyển tiền 2PC.
    \item Tra cứu trạng thái giao dịch theo tx\_id.
    \item Xem danh sách giao dịch gần nhất.
    \item Chạy recovery thủ công.
\end{itemize}

\section{Chiến lược phân mảnh}

\subsection{Horizontal fragmentation}
{Bảng accounts} được phân bổ theo site ngân hàng. Điều kiện phân mảnh nghiệp vụ có thể biểu diễn:
\begin{itemize}
    \item Fragment F\textsubscript{A}: tài khoản thuộc Bank A (lưu tại bank1).
    \item Fragment F\textsubscript{B}: tài khoản thuộc Bank B (lưu tại bank2).
    \item Fragment F\textsubscript{C}: tài khoản thuộc Bank C (lưu tại bank3).
\end{itemize}

{Predicate ví dụ (mức demo):}
\begin{itemize}
    \item account\_number bắt đầu bằng nhóm số của từng chi nhánh/ngân hàng.
    \item Hoặc ánh xạ theo bảng config site thay vì hard-code predicate trong SQL.
\end{itemize}

\subsection{Vertical fragmentation (có chủ đích)}
Không chia cột theo nghĩa truyền thống trên cùng một bảng, nhưng về kiến trúc:
\begin{itemize}
    \item Dữ liệu vận hành giao dịch tập trung log ở bank1: transaction\_log, idempotency\_keys.
    \item Dữ liệu tài khoản gốc nằm tại từng site participant.
\end{itemize}

\subsection{Hybrid fragmentation}
Hệ thống hiện tại có thể xem là {hybrid practical}:
\begin{itemize}
    \item Horizontal cho dữ liệu tài khoản theo site.
    \item Tách lớp metadata giao dịch (log/idempotency) về site coordinator.
\end{itemize}

\section{Giải thích lựa chọn thiết kế}
\subsection{Tại sao phân theo chi nhánh/site ngân hàng?}
\begin{itemize}
    \item Phân bổ đúng theo ownership dữ liệu (mỗi ngân hàng sở hữu số dư tài khoản của mình).
    \item Giảm độ phụ thuộc chéo và giảm tải cho một CSDL tập trung.
    \item Gần với bài toán thực tế liên ngân hàng.
\end{itemize}

\subsection{Tại sao không replicate full toàn bộ accounts?}
\begin{itemize}
    \item Replicate full làm tăng chi phí đồng bộ và rủi ro conflict.
    \item Bài toán chính cần đảm bảo atomicity giao dịch, không phải sao chép toàn bộ dữ liệu read-heavy.
    \item 2PC + log/recovery đã đáp ứng mục tiêu môn học rõ ràng hơn replication toàn phần.
\end{itemize}

    \item Chỉ commit khi tất cả participant prepare thành công.
\end{itemize}

\subsection{Xử lý conflict}
\begin{itemize}
    \item Conflict write được giảm nhờ lock cấp transaction XA.
    \item Conflict yêu cầu lặp lại (double submit) được xử lý bằng idempotency\_keys.
    \item Trường hợp lệch pha COMMIT\_A được xử lý bằng compensation/recovery.
\end{itemize}

\begin{itemize}

\end{itemize}

\chapter{Xây dựng và Demo}

\section{Store Procedure phân tán (tương đương trong project hiện tại)}
Project hiện tại không viết SP SQL thuần cho toàn bộ luồng 2PC mà triển khai ở tầng ứng dụng (\texttt{execute\_transfer()} trong backend). Về mặt thuật toán, nó tương đương một stored-procedure phân tán:

\begin{lstlisting}[language=Pseudo,caption={Giả mã xử lý giao dịch 2PC cấp ứng dụng}]
begin transfer_2pc(from_acc, to_acc, amount, description)
    create xid, tx_id
    log_phase(PREPARING)

    parallel:
        prepare participant A
        prepare participant B
    wait with timeout

    if any participant fail or timeout then
        rollback all
        log_phase(ABORTED or TIMEOUT)
        return error
    end if

    log_phase(PREPARED)
    log_phase(COMMITTING)

    commit A
    log_phase(COMMIT_A)

    commit B
    log_phase(COMMITTED)
    save receipt
    return success

on phase-2 partial failure:
    rollback participant chua commit
    do compensation cho participant da commit
    return error with partial_failure
end
\end{lstlisting}
